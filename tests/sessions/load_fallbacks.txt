# Load fallbacks (SCENE-LOAD-FAIL-FALLBACK-67, DBG-TP-NOT-FOUND-BOX-67; docs/LOAD_ERRORS.md): every refusal and every
# load-failure kind shows its own OK box, the player stays or is put back where he was, and the game runs on. Run as a
# scripted session from a place in play (a full image's start map, or a save; the loads go to Seyda Neen and back),
# 32 MB Fast, warp off, the position overlay on: each box is pictured before OK (Enter), and the map is read before and
# after. The doors and the travel services reach the same catcher (tests/aga_scene_test.c covers a door's failed load
# and a ride refused before its fare). The "emu" line runs the emulator's folder drives on its own thread
# (CHIM-SEYDA-BALMORA-LOAD-HANG-43): the remote console's folder requests then never stop a test session.
emu deterministic=1
wait 8
cmd aw_crosshair off
cmd dbg all on
cmd dbg headlamp off
state map
shot lf0-start-in-play
# LF1: a name that is no map; nothing is left
cmd dbg tp nowhere
wait 8
shot lf1-dbg-tp-nowhere-not-found-box
key Return
wait 2
state map
# LF3: a point no map holds (the search of the disk's maps takes a few seconds: 3 s missed the box on 0d6c6e87)
cmd dbg tp 1900000 -1900000
wait 15
shot lf3-dbg-tp-point-no-map-box
key Return
wait 2
state map
# LF9 and the way back: the next load fails (dbg loadfail 1)
cmd dbg loadfail 1
cmd dbg tp seydaneen
wait 90
state map
shot lf9-forced-failure-way-back-box
key Return
wait 3
shot lf9-after-ok-back-where-he-was
# LF4 bad lump: the map's first section read gets bad data (a short read and a bad header: tests/aga_bsp_slice_load_test.c)
cmd dbg loadfail 1 lump
cmd dbg tp seydaneen
wait 90
state map
shot lf4-bad-lump-way-back-box
key Return
wait 2
# LF5: an allocation of the load fails
cmd dbg loadfail 1 memory
cmd dbg tp seydaneen
wait 90
state map
shot lf5-not-enough-memory-way-back-box
key Return
wait 2
# LF7: the load never finishes; a time limit of 10 s for the test
cmd dbg loadfail timeout 10
cmd dbg loadfail 1 hang
cmd dbg tp seydaneen
wait 100
state map
shot lf7-load-did-not-finish-way-back-box
key Return
wait 2
cmd dbg loadfail timeout 120
# the game runs on: a real load goes through
cmd dbg tp seydaneen
wait 90
state map
shot lf-real-load-seyda-after-the-failures
# the way back fails as well (dbg loadfail 2): the main menu and the load menu name the error
cmd dbg loadfail 2
cmd dbg tp balmora
wait 150
state map
shot lf9-way-back-failed-load-menu-notice
